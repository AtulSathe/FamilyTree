using System.Text.Json;
using FamilyTree.Api.Data;
using FamilyTree.Api.Domain;

namespace FamilyTree.Api.Services;

public class AuditService(AppDbContext db)
{
    public async Task LogAsync(
        Guid? userId, string entityType, Guid entityId,
        string action, object? oldValue, object? newValue)
    {
        db.AuditLog.Add(new AuditLog
        {
            UserId     = userId,
            EntityType = entityType,
            EntityId   = entityId,
            Action     = action,
            OldValue   = oldValue is null ? null : JsonSerializer.Serialize(oldValue),
            NewValue   = newValue is null ? null : JsonSerializer.Serialize(newValue),
            ChangedAt  = DateTime.UtcNow
        });
        await db.SaveChangesAsync();
    }
}

public class SearchService(AppDbContext db)
{
    public async Task<List<SearchResultDto>> SearchAsync(string q, string? surname)
    {
        var query = db.PersonTreeMemberships
            .Include(m => m.Person)
            .Include(m => m.FamilyTree)
            .Where(m => EF.Functions.Like(m.Person.FullName, $"%{q}%"));

        if (!string.IsNullOrWhiteSpace(surname))
            query = query.Where(m => m.FamilyTree.Surname == surname);

        return await query
            .Select(m => new SearchResultDto(
                m.PersonId, m.Person.FullName, m.Person.Location,
                m.Person.BirthMonthYear, m.Person.PhotoBlobUrl,
                m.FamilyTreeId, m.FamilyTree.Surname))
            .ToListAsync();
    }
}

public class SurnameRelationshipService(AppDbContext db)
{
    public async Task<List<SurnameResponse>> ListAllSurnamesAsync() =>
        await db.FamilyTrees
            .Select(t => new SurnameResponse(t.Surname, t.Id, t.PersonMemberships.Count))
            .ToListAsync();

    public async Task<List<SurnameLinkResponse>> ListCrossLinksAsync() =>
        await db.SurnameLinks
            .Include(s => s.LinkPerson)
            .Select(s => new SurnameLinkResponse(
                s.SurnameA, s.SurnameB, s.LinkPersonId,
                s.LinkPerson.FullName, s.RelationshipLevel))
            .ToListAsync();

    public async Task<PersonResponse?> GetRecentLivingMemberAsync(string surname) =>
        await db.PersonTreeMemberships
            .Where(m => m.FamilyTree.Surname == surname && m.Person.DeathMonthYear == null)
            .OrderByDescending(m => m.Person.BirthMonthYear)
            .Select(m => new PersonResponse(
                m.Person.Id, m.Person.FullName, m.Person.NameBefore,
                m.Person.Phone, m.Person.Location, m.Person.BirthMonthYear,
                m.Person.DeathMonthYear, m.Person.PhotoBlobUrl, m.Person.PrimaryTreeId))
            .FirstOrDefaultAsync();
}

public class UserService(AppDbContext db, AuditService audit)
{
    public async Task<bool> UpdateRoleAsync(Guid id, string role, Guid actorId)
    {
        var user = await db.Users.FindAsync(id);
        if (user is null) return false;
        var oldRole = user.Role;
        user.Role = role;
        await db.SaveChangesAsync();
        await audit.LogAsync(actorId, "User", id, "UPDATE",
            new { Role = oldRole }, new { Role = role });
        return true;
    }

    public async Task AssignAdminAsync(Guid treeId, Guid userId)
    {
        var existing = await db.FamilyTreeAdmins.FindAsync(treeId, userId);
        if (existing is not null) return;
        db.FamilyTreeAdmins.Add(new FamilyTreeAdmin { FamilyTreeId = treeId, UserId = userId });
        await db.SaveChangesAsync();
    }
}

public async Task<List<RelationDto>> GetNeighboursAsync(
    string personId,
    Func<IEnumerable<string>, Task<Dictionary<string, (string fullName, string? photoUrl)>>> sqlLookup)
{
    // Query outgoing edges — "I am the source"
    var outEdges = await graph.QueryAsync(
        "g.V().has('person','personId',pid).outE().project('type','otherId')"
        + ".by(label()).by(inV().values('personId'))",
        new Dictionary<string, object> { ["pid"] = personId });

    // Query incoming edges — "I am the target"
    var inEdges = await graph.QueryAsync(
        "g.V().has('person','personId',pid).inE().project('type','otherId')"
        + ".by(label()).by(outV().values('personId'))",
        new Dictionary<string, object> { ["pid"] = personId });

    // Bulk SQL fetch — one query for all neighbour names
    var allIds = outEdges.Select(e => e["otherId"]?.ToString()!)
        .Concat(inEdges.Select(e => e["otherId"]?.ToString()!))
        .Where(id => id is not null).Distinct().ToList();

    var sqlData = await sqlLookup(allIds);
    // ... map to RelationDto list with direction logic
}

public async Task<PersonResponse> CreateAsync(CreatePersonRequest req, Guid actorId)
{
    var person = new Person { FullName = req.FullName, PrimaryTreeId = req.TreeId, ... };
    db.Persons.Add(person);
    db.PersonTreeMemberships.Add(new PersonTreeMembership
        { PersonId = person.Id, FamilyTreeId = req.TreeId, Role = req.Role });
    await db.SaveChangesAsync();
    await graph.AddPersonVertexAsync(person.Id.ToString(), person.FullName, ...);
    await audit.LogAsync(...);
    return new PersonResponse(...);
}

public async Task<PhotoUploadUrlResponse?> GetPhotoUploadUrlAsync(Guid id)
{
    var connStr       = config["AzureBlob:ConnectionString"];
    var containerName = config["AzureBlob:ContainerName"] ?? "person-photos";
    var blobName      = $"{id}/{Guid.NewGuid()}.jpg";

    var blobClient = new BlobServiceClient(connStr)
        .GetBlobContainerClient(containerName)
        .GetBlobClient(blobName);

    var sasUri = blobClient.GenerateSasUri(
        BlobSasPermissions.Write | BlobSasPermissions.Create,
        DateTimeOffset.UtcNow.AddMinutes(15));

    return new PhotoUploadUrlResponse(sasUri.ToString(), blobName);
}

public async Task AddRelationshipAsync(Guid treeId, CreateRelationshipRequest req, Guid actorId)
{
    // Validate membership
    var memberIds = await db.PersonTreeMemberships
        .Where(m => m.FamilyTreeId == treeId
                 && (m.PersonId == req.PersonAId || m.PersonId == req.PersonBId))
        .Select(m => m.PersonId).ToListAsync();

    if (!memberIds.Contains(req.PersonAId) || !memberIds.Contains(req.PersonBId))
        throw new InvalidOperationException("Both persons must be members of the tree.");

    // Primary edge A → B
    await graph.AddEdgeAsync(aId, bId, req.RelationshipType);

    // Reverse edge: directed types get logical inverse, undirected get same label
    var reverseMap = new Dictionary<string, string>
    {
        ["parent_of"]          = "child_of",
        ["child_of"]           = "parent_of",
        ["step_parent_of"]    = "child_of",
        ["adoptive_parent_of"] = "child_of"
    };
    if (reverseMap.TryGetValue(req.RelationshipType, out var reverse))
        await graph.AddEdgeAsync(bId, aId, reverse);
    else
        await graph.AddEdgeAsync(bId, aId, req.RelationshipType);
}

🧠 Design Decisions

▼
DecisionReasoning
Gremlin queries use project() to return IDictionary GraphDbContext.QueryAsync is typed as IReadOnlyCollection<IDictionary>. Queries returning raw scalar values (e.g. .values('personId')) would cause a cast failure at runtime. Using project().by() always returns a map, matching the declared return type. No changes to GraphDbContext were needed.
sqlLookup delegate in GetNeighboursAsync GraphTraversalService must not take AppDbContext as a dependency (graph and SQL are separate concerns). Injecting a delegate lets callers provide the SQL lookup, keeping the service focused on Gremlin. The result is a single bulk SQL query per Gremlin traversal — no N+1.
Partial update in PersonService.UpdateAsync — null = skip PUT body that omits a field should not overwrite existing data. All nullable request fields are checked: if (req.Field is not null) person.Field = req.Field. This is safe because the domain fields are all nullable strings.
PersonService.DeleteAsync removes memberships before the person PersonTreeMembership → FamilyTree FK uses DeleteBehavior.Restrict, so deleting a Person while memberships exist will fail. Memberships are removed first, then the person.
Reverse edges stored in Cosmos for directed relationships CLAUDE.md explicitly states "both directions stored" for parent_of/child_of edges. The reverseMap encodes each directed type's inverse. Undirected types (spouse, sibling_of, in_law_of) get the same label repeated in both directions — this matches the seed.groovy pattern.
SAS token includes BlobSasPermissions.Create in addition to Write For a new blob upload (not updating an existing one) the Create permission is required alongside Write. Without it, the PUT from the browser would return 403.
Primary constructor injection (class Foo(Dep dep)) .NET 8 primary constructors remove the need for private fields and constructor bodies for DI. All services use this pattern for brevity and alignment with the CLAUDE.md convention of minimal boilerplate.

⚠️ Assumptions Made

▼
AssumptionRisk / Mitigation
Gremlin project().by(inV().values('personId')) works on TinkerPop 3.7.1 This is standard TinkerPop syntax. Verified against the Gremlin spec. Risk: low. Will be confirmed when Docker stack is running in integration tests (Phase D).
OrderByDescending(BirthMonthYear) as a string is lexicographic, not chronological "Jan 1990" vs "Dec 1985" — string sort gives wrong order. Risk: low for MVP (only affects the "most recent living member" query for sidebar centering). A proper fix would parse the month/year and sort numerically. Deferred to future iteration.
Photo file extension is always .jpg in the SAS blob name FR-14 doesn't specify formats. The SAS URL covers that specific blob name. The client must PUT a JPEG, or the extension is misleading. A future improvement would accept the content type as a parameter. Low risk for phase 1.
Duplicate edge prevention not implemented in AddEdgeAsync If called twice with the same person pair + edge label, Gremlin will add duplicate edges. This is intentional — prevention is the caller's responsibility (endpoints will check). TinkerPop supports duplicate edges by design.
