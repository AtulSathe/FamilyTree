💻 AuthHelper.cs

▼
internal static class AuthHelper
{
    internal static Guid? GetUserId(HttpContext ctx) =>
        Guid.TryParse(ctx.User.FindFirstValue(ClaimTypes.NameIdentifier), out var id) ? id : null;

    internal static string GetRole(HttpContext ctx) =>
        ctx.User.FindFirstValue("extension_Role") ?? string.Empty;

    internal static bool IsPowerAdmin(HttpContext ctx) =>
        GetRole(ctx) == "power_admin";

    internal static bool CanMutate(HttpContext ctx) =>
        GetRole(ctx) is "power_admin" or "family_admin";

    internal static bool CanEditTree(HttpContext ctx, Guid treeId)
    {
        var role = GetRole(ctx);
        if (role == "power_admin") return true;
        if (role != "family_admin") return false;
        var assigned = ctx.User.FindFirstValue("extension_AssignedTrees") ?? string.Empty;
        return assigned.Split(',', StringSplitOptions.RemoveEmptyEntries)
                       .Contains(treeId.ToString());
    }
}

💻 PersonEndpoints.cs — Update (cross-tree ownership check)

▼
private static async Task<IResult> Update(
    Guid id, UpdatePersonRequest req,
    HttpContext ctx, PersonService svc, AppDbContext db)
{
    if (!AuthHelper.CanMutate(ctx)) return AuthHelper.Forbid();
    var actorId = AuthHelper.GetUserId(ctx);
    if (actorId is null) return AuthHelper.Unauthorized();

    // family_admin: verify actor manages ≥1 tree this person belongs to
    if (!AuthHelper.IsPowerAdmin(ctx))
    {
        var assigned = (ctx.User.FindFirst("extension_AssignedTrees")?.Value ?? string.Empty)
            .Split(',', StringSplitOptions.RemoveEmptyEntries);
        var personTrees = await db.PersonTreeMemberships
            .Where(m => m.PersonId == id)
            .Select(m => m.FamilyTreeId.ToString())
            .ToListAsync();
        if (!personTrees.Any(t => assigned.Contains(t)))
            return AuthHelper.Forbid();
    }
    var result = await svc.UpdateAsync(id, req, actorId.Value);
    return result is null ? Results.NotFound() : Results.Ok(result);
}

💻 TreeEndpoints.cs — AddRelationship (error mapping)

▼
private static async Task<IResult> AddRelationship(
    Guid treeId, CreateRelationshipRequest req,
    HttpContext ctx, TreeService svc)
{
    if (!AuthHelper.CanEditTree(ctx, treeId)) return AuthHelper.Forbid();
    var actorId = AuthHelper.GetUserId(ctx);
    if (actorId is null) return AuthHelper.Unauthorized();
    try
    {
        await svc.AddRelationshipAsync(treeId, req, actorId.Value);
        return Results.Created($"/api/v1/trees/{treeId}/relationships", null);
    }
    catch (InvalidOperationException ex)
    {
        return Results.BadRequest(new { error = ex.Message });
    }
}

private static async Task<IResult> Search(
    string? q, string? surname, SearchService svc)
{
    if (string.IsNullOrWhiteSpace(q))
        return Results.BadRequest(new { error = "Query parameter 'q' is required." });
    return Results.Ok(await svc.SearchAsync(q, surname));
}

💻 AdminEndpoints.cs — Audit Log (paginated)

▼
private static async Task<IResult> GetAuditLog(
    HttpContext ctx, AppDbContext db,
    int page = 1, int pageSize = 50)
{
    if (!AuthHelper.IsPowerAdmin(ctx)) return AuthHelper.Forbid();

    pageSize = Math.Clamp(pageSize, 1, 200);
    page     = Math.Max(page, 1);

    var total = await db.AuditLog.CountAsync();
    var items = await db.AuditLog
        .OrderByDescending(a => a.ChangedAt)
        .Skip((page - 1) * pageSize)
        .Take(pageSize)
        .Select(a => new AuditLogResponse(...))
        .ToListAsync();

    return Results.Ok(new { total, page, pageSize, items });
}

🧠 Design Decisions

▼
DecisionReasoning
AuthHelper as an internal static class in the Endpoints namespace Auth logic is endpoint-specific (claim reading, HTTP result helpers). Making it internal prevents it being misused in services. Static avoids DI registration for a utility that has no state.
Manual claim-based auth instead of [Authorize] attributes MockAuthMiddleware injects a ClaimsPrincipal without going through ASP.NET Core's authorization middleware. Using [Authorize] would require the real middleware to be wired even in dev. Manual checks are also more explicit and testable — you can read exactly what is being checked in the handler code.
Route registration order: search-existing before /{id} in PersonEndpoints Minimal API routing is first-match-wins for ambiguous segments. A request to /persons/search-existing could match /{id} and attempt to parse "search-existing" as a Guid (which fails as a 400, not a 404). Registering the static path first avoids this entirely. Same reasoning applied to /surnames/relationships before /{surname}/recent.
PUT /persons/{id} does a DB query for family_admin tree ownership check Unlike POST /persons (where treeId is in the request body) or tree endpoints (where treeId is in the URL), a person edit has no treeId in the URL. The only correct check is: does this person belong to any tree the admin manages? One lightweight SELECT FamilyTreeId FROM PersonTreeMemberships WHERE PersonId = ? suffices.
Dropped MapRelationshipEndpoints() from Program.cs The original plan stub referenced RelationshipEndpoints, but relationship operations are logically tree-scoped (you add a relationship within a tree context) and were implemented as POST /trees/{treeId}/relationships in TreeEndpoints. No separate file needed.
Audit log returns {total, page, pageSize, items} envelope A paginated endpoint without a total count forces clients to guess when pagination ends. Including total allows the frontend to render a "page X of Y" indicator or stop requesting pages when none remain.

⚠️ Assumptions Made

▼
AssumptionRisk / Mitigation
extension_AssignedTrees is a single GUID string (not comma-separated) in mock auth MockAuthMiddleware adds only one tree ID. The code splits on ',' so it handles single or multiple values. If Azure AD B2C returns multiple trees as comma-separated, it still works. Low risk.
Minimal API parameter binding for query strings works without [FromQuery] In .NET 8 minimal APIs, simple types in method signatures are automatically bound from query strings. Confirmed by .NET docs. If it fails in testing, adding [FromQuery] is a one-line fix.
role in AddMember is a query parameter (not in the body) The route is POST /trees/{treeId}/members/{personId} with no request body. Role is a query param defaulting to "member". This is convenient but means the role can't be sent in a JSON body. If this becomes awkward for the frontend, a small request body DTO can be added later.
